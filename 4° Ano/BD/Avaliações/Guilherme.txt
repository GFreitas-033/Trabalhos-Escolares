-- 1- Liste o número do pedido e o valor total de cada pedido (orders e orderdetails), exibindo apenas os pedidos cujo valor total seja superior a 50.000.
SELECT o.orderNumber, SUM(od.quantityOrdered * od.priceEach) AS valor_total
FROM orders o
INNER JOIN orderdetails od ON o.orderNumber = od.orderNumber
GROUP BY o.orderNumber
HAVING valor_total > 50000
ORDER BY valor_total ASC;

-- 4- Crie uma função chamada TotalPagoCliente que receba o número do cliente e retorne o valor total de pagamentos desse cliente. Em seguida, use essa função para listar o nome de todos os
-- clientes e o total que cada um já pagou.
DELIMITER $$
CREATE FUNCTION TotalPagoCliente(p_customerNumber INT)
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
	DECLARE resultado DECIMAL(10,2);
    
    SELECT SUM(p.amount) AS valor_total
    INTO resultado
    FROM payments p
    GROUP BY p.customerNumber
    HAVING p.customerNumber = p_customerNumber;
    
    RETURN resultado;
END $$
DELIMITER ;

-- 5- Crie uma função chamada CategoriaPedido que receba um número de pedido e retorne uma classificação textual:
-- "PEQUENO" se o valor do pedido for menor que 10.000,
-- "MÉDIO" se for entre 10.000 e 50.000,
-- "GRANDE" se for acima de 50.000.
-- Liste os pedidos e sua categoria.
DELIMITER $$
CREATE FUNCTION CategoriaPedido(p_orderNumber INT)
RETURNS varchar(255)
DETERMINISTIC
BEGIN
    DECLARE total decimal(10,2);
    
    SELECT SUM(od.quantityOrdered * od.priceEach) AS valor_total
    INTO total
    FROM orderdetails od
    GROUP BY od.orderNumber
    HAVING od.orderNumber = p_orderNumber;
    
	IF total < 10000 THEN
        RETURN 'PEQUENO';
	END IF;
	IF total > 10000 AND total < 50000 THEN
        RETURN 'MÉDIO';
	END IF;
	IF total > 50000 THEN
        RETURN 'GRANDE';
	END IF;
END $$
DELIMITER ;

SELECT o.orderNumber, CategoriaPedido(o.orderNumber) AS categoria FROM orders o;
DROP FUNCTION CategoriaPedido;
